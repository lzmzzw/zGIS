cmake_minimum_required(VERSION 3.16)
project(zgis_sqlite C)
if(NOT SQLITE_SOURCE)
  message(FATAL_ERROR "SQLITE_SOURCE is required")
endif()
add_library(sqlite3 STATIC "${SQLITE_SOURCE}/sqlite3.c")
target_include_directories(sqlite3 PUBLIC "${SQLITE_SOURCE}")
target_compile_definitions(sqlite3 PRIVATE SQLITE_ENABLE_COLUMN_METADATA SQLITE_ENABLE_RTREE SQLITE_THREADSAFE=1)
set_target_properties(sqlite3 PROPERTIES POSITION_INDEPENDENT_CODE ON)
add_executable(sqlite3_cli "${SQLITE_SOURCE}/shell.c")
set_target_properties(sqlite3_cli PROPERTIES OUTPUT_NAME sqlite3)
target_link_libraries(sqlite3_cli PRIVATE sqlite3 ${CMAKE_DL_LIBS})
if(UNIX)
  target_link_libraries(sqlite3_cli PRIVATE m pthread)
endif()
install(TARGETS sqlite3 sqlite3_cli ARCHIVE DESTINATION lib RUNTIME DESTINATION bin)
install(FILES "${SQLITE_SOURCE}/sqlite3.h" "${SQLITE_SOURCE}/sqlite3ext.h" DESTINATION include)
